        let adTimerInterval;
let adSecondsLeft = 30;
let isAdWatched = false;

function watchMonetagAd() {
  const watchBtn = document.getElementById('watchAdBtn');
  const timerText = document.getElementById('adTimer');

  if (isAdWatched) {
    // Agar ad dekh liya hai toh reward claim/credit karne ki API call lagayein
    claimAdRewardFromBackend();
    return;
  }

  // Button disable karein aur timer start karein
  watchBtn.disabled = true;
  watchBtn.classList.add('opacity-50', 'cursor-not-allowed');
  adSecondsLeft = 30;

  // Optional: Agar Monetag/Ad ka direct link open karna ho
  // window.open('YOUR_AD_LINK_HERE', '_blank');

  adTimerInterval = setInterval(() => {
    adSecondsLeft--;
    watchBtn.innerText = `Watching... (${adSecondsLeft}s)`;
    timerText.innerText = `Please wait ${adSecondsLeft}s to unlock reward`;

    if (adSecondsLeft <= 0) {
      clearInterval(adTimerInterval);
      isAdWatched = true;
      watchBtn.disabled = false;
      watchBtn.classList.remove('opacity-50', 'cursor-not-allowed');
      watchBtn.innerText = 'Claim 100 Coins 🪙';
      watchBtn.classList.remove('bg-amber-400', 'hover:bg-amber-300');
      watchBtn.classList.add('bg-emerald-500', 'hover:bg-emerald-400');
      timerText.innerText = 'Ad complete! Click the button above to claim.';
    }
  }, 1000);
}

function claimAdRewardFromBackend() {
  const currentUserId = window.Telegram?.WebApp?.initDataUnsafe?.user?.id || "7985106214";

  fetch('/api/bot', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      action: 'claim_ad_reward',
      userId: currentUserId,
      watchDuration: 30
    })
  })
  .then(res => res.json())
  .then(data => {
    if (data.success) {
      alert("🎉 " + data.message);
      document.getElementById('total-balance').innerText = `${data.newBalance} Coins`;
      
      // Reset button state for next ad
      isAdWatched = false;
      const watchBtn = document.getElementById('watchAdBtn');
      watchBtn.innerText = 'Watch Ad Now';
      watchBtn.classList.remove('bg-emerald-500', 'hover:bg-emerald-400');
      watchBtn.classList.add('bg-amber-400', 'hover:bg-amber-300');
      document.getElementById('adTimer').innerText = '';
    } else {
      alert("Error: " + data.error);
    }
  })
  .catch(err => alert("Something went wrong!"));
}

    // WITHDRAWAL TIMING VALIDATION (MON-FRI, 8AM-6PM IST)
    function isWithdrawalTimeValid() {
      const now = new Date();
      const utc = now.getTime() + (now.getTimezoneOffset() * 60000);
      const istTime = new Date(utc + (3600000 * 5.5));

      const day = istTime.getDay(); 
      const hours = istTime.getHours();

      const isWeekday = day >= 1 && day <= 5; 
      const isWithinHours = hours >= 8 && hours < 18; 

      return isWeekday && isWithinHours;
    }

      const bgClass = type === 'error' ? 'bg-red-600 border-2 border-slate-900' : type === 'success' ? 'bg-emerald-600 border-2 border-slate-900' : 'bg-slate-900 border-2 border-slate-900';

      toast.className = `${bgClass} text-white px-4 py-3 rounded-xl shadow-xl text-xs font-black flex items-center justify-between toast-animate`;
      toast.innerHTML = `<span>${message}</span><button onclick="this.parentElement.remove()" class="ml-3 text-sm font-black">&times;</button>`;

      container.appendChild(toast);
      setTimeout(() => {
        if (toast.parentElement) toast.remove();
      }, 3500);
    }
  </script>
